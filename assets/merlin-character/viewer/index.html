Use the screens from code

Each surface accepts a canvas, image, or video element. The content moves with the panel and its fragments.

// Show a reusable 5:6 panel design.
merlin.showPanel('timer', {}, 'right');

const screen = merlin.screens.right;
screen.draw((ctx, width, height) => {
  ctx.fillStyle = '#ffffff';
  ctx.fillRect(0, 0, width, height);
  ctx.fillStyle = '#172a3b';
  ctx.font = '56px sans-serif';
  ctx.fillText('Hello, MERLIN.', 60, 110);
});
merlin.go('right');

// Sit between two independently editable screens.
merlin.go('both');
merlin.dismissScreen('left'); // keep the right screen

merlin.go('ceiling');
merlin.dockScreen('left'); // carry its content to the left panel
